# 1 数组

## 1.1 一维数组与二维数组

**定义：**数组是一种**连续存储结构**，通过下标进行随机访问。

**特点：**

- 随机访问
- 连续内存
- 固定大小

## 1.2 动态数组

**扩容机制：**

1. 扩容策略：
   - Java ArrayList：1.5倍扩容
   - C++ vector：2倍扩容
   - Python list：1.125倍扩容 + 常数
2. 扩容过程：
   - 分配新的更大的连续内存空间
   - 将旧数组元素逐个复制到新数组
   - 释放旧数组内存
   - 更新指针指向新数组

## 1.3 时间复杂度

| 操作        | 时间复杂度 |
| :---------- | :--------- |
| 访问 arr[i] | O(1)       |
| 查找元素    | O(n)       |
| 头部插入    | O(n)       |
| 尾部插入    | O(1)       |
| 删除元素    | O(n)       |
| 排序        | O(nlogn)   |

## 1.4 面试技巧

### 1.4.1 双指针

**常见形式：**

1. **左右指针**

   - 用途：有序数组找两个数

   - 模板：
   
     ```c++
     int left = 0;
     int right = nums.size()-1;
     
     while(left < right)
     {
         if(condition)
             left++;
         else
             right--;
     }
     ```
   
   - 典型题目：
   
     - [两数之和](https://leetcode.cn/problems/two-sum/)
   
       ```c++
       class Solution {
       public:
           vector<int> twoSum(vector<int>& nums, int target) {
               // 建立一个数组作为结果输出
               vector<int> res;
               
               // 将数组下标与数值建立二元组，避免排序后下标乱序
               vector<pair<int,int>> index_nums;
               for(int i=0; i<=nums.size()-1; i++)
               {
                   pair<int,int> p = {nums[i],i};
                   index_nums.push_back(p);
               }
       
               // 排序
               sort(index_nums.begin(),index_nums.end());
       
               // 左右指针
               int left = 0;
               int right = nums.size() - 1;
               while(left < right)
               {
                   if(index_nums[left].first + index_nums[right].first < target)
                   {
                       left++;
                   }
                   else if(index_nums[left].first + index_nums[right].first > target)
                   {
                       right--;
                   }
                   else
                   {
                       res.push_back(index_nums[left].second);
                       res.push_back(index_nums[right].second);
                       break;
                   }
               }
       
               return res;
           }
       };
       ```
   
     - [三数之和](https://leetcode.cn/problems/3sum/)
   
       ```c++
       class Solution {
       public:
           vector<vector<int>> threeSum(vector<int>& nums) {
               // 建立一个数组作为结果输出
               vector<vector<int>> res;
       
               // 自动去重和排序
               sort(nums.begin(),nums.end());
       
               // 左右指针
               for(int i=0; i<=static_cast<int>(nums.size())-3; i++)
               {
                   int left = i+1;
                   int right = nums.size()-1;
       
                   // 索引i去重
                   if(i > 0 && nums[i] == nums[i - 1])
                       continue;
       
                   while(left<right)
                   {
                       if(nums[i] + nums[left] + nums[right] < 0)
                       {
                           left++;
                       }
                       else if(nums[i] + nums[left] + nums[right] > 0)
                       {
                           right--;
                       }
                       else
                       {
                           res.push_back({nums[i], nums[left], nums[right]});
                           
                           // 索引left和right去重
                           while(left<right && nums[left]==nums[left+1])
                           {
                               left++;
                           }
                           while(left<right && nums[right]==nums[right-1])
                           {
                               right--;
                           }
                           left++;
                           right--;
                       }
                   }
               }
       
               return res;
           }
       };
       ```
   
     - [盛最多水的容器](https://leetcode.cn/problems/container-with-most-water/)
   
       ```c++
       class Solution {
       public:
           int maxArea(vector<int>& height) {
               // 最大面积
               int max_area = INT_MIN;
       
       
               // 左右指针
               int left = 0;
               int right = height.size()-1;
               while(left < right)
               {
                   int area = (right - left) * min(height[left],height[right]);
                   max_area = max(max_area,area);
       
                   if(height[left] < height[right])
                   {
                       left++;
                   }
                   else
                   {
                       right--;
                   }
               }
       
               return max_area;
           }
       };
       ```
   
2. **快慢指针**

   - 用途：删除/覆盖元素

   - 模板：
   
     ```c++
     int slow = 0;
     
     for(int fast = 0; fast < nums.size(); fast++)
     {
         if(condition)
         {
             nums[slow] = nums[fast];
             slow++;
         }
     }
     ```
   
   - 典型题目：
   
     - [删除有序数组重复元素](https://leetcode.cn/problems/remove-duplicates-from-sorted-array/)
   
       ```c++
       class Solution {
       public:
           int removeDuplicates(vector<int>& nums) {
               // 快慢指针
               int slow = 0;
               for(int fast=0; fast<=nums.size()-1; fast++)
               {
                   if(nums[slow]!=nums[fast])
                   {
                       slow++;
                       nums[slow] = nums[fast];
                   }
               }
       
               return slow+1;
           }
       };
       ```
   
     - [移动零](https://leetcode.cn/problems/move-zeroes/)
   
       ```c++
       class Solution {
       public:
           void moveZeroes(vector<int>& nums) {
               // 快慢指针
               int slow = 0;
               for(int fast=0; fast<=nums.size()-1; fast++)
               {
                   if(nums[fast]!=0)
                   {
                       nums[slow] = nums[fast];
                       slow++;
                   }
               }
       
               // 将后续的元素置为0
               while(slow <= nums.size()-1)
               {
                   nums[slow] = 0;
                   slow++;
               }
       
               return;
           }
       };
       ```
   
3. **滑动窗口**

   - 用途：连续子数组

   - 模板：
   
     ```c++
     int left = 0;
     
     for(int right=0; righ<n; right++)
     {
         while(condition)
         {
             left++;
         }
         
         ans = max(ans, right-left+1);
     }
     ```
   
   - 典型题目：
   
     - [长度最小的子数组](https://leetcode.cn/problems/2VG8Kg/description/)
   
       ```c++
       class Solution {
       public:
           int minSubArrayLen(int target, vector<int>& nums) {
               // 滑动窗口
               int left = 0;
               int sum = 0;
               int ans = INT_MAX;
               for(int right=0; right<=nums.size()-1; right++)
               {
                   sum += nums[right];
       
                   while(sum>=target)
                   {
                       ans = min(ans,right-left+1);
                       sum -= nums[left];
                       left++;
                   }
               }
       
               return ans == INT_MAX? 0 : ans;
           }
       };
       ```

### 1.4.2 前缀和

用途：区间求和

模板：

```c++
vector<int> prefix(n+1);

for(int i=0; i<n; i++)
{
    prefix[i+1] = prefix[i] + nums[i];
}
```

典型题目：

- [和为K的子数组](https://leetcode.cn/problems/QTMn0o/)

  ```c++
  class Solution {
  public:
      int subarraySum(vector<int>& nums, int k) {
          // 前缀和
          unordered_map<int,int> mp;
          mp[0] = 1;
          int ans = 0;
          int sum = 0;
  
          for(int i=0; i<=nums.size()-1; i++)
          {
              sum += nums[i];
  
              if(mp.count(sum-k))
              {
                  ans += mp[sum-k];
              }
  
              mp[sum]++;
          }
  
          return ans;
      }
  };
  ```

### 1.4.3 二分查找

int left = 0;
int right = n-1;

while(left<=right)

    if(nums[mid]==target)
        return mid;
    else if(nums[mid]<target)
        left = mid + 1;
    else
        right = mid - 1;
}

典型题目：

- [搜索旋转排序数组](https://leetcode.cn/problems/search-in-rotated-sorted-array/z)

  ```c++
  class Solution {
  public:
      int search(vector<int>& nums, int target) {        
          // 二分查找
          int left = 0;
          int right = nums.size()-1;
  
          while(left<=right)
          {
              int mid = left + (right - left) / 2;
  
              if(nums[mid] == target)
              {
                  return mid;
              }
              // 左区间有序
              else if(nums[left] <= nums[mid])
              {
                  if(nums[left] <= target && target < nums[mid])
                  {
                      right = mid - 1;
                  }
                  else
                  {
                      left = mid + 1;
                  }
              }
              // 右区间有序
              else
              {
                  if(nums[mid] < target && target <= nums[right])
                  {
                      left = mid + 1;
                  }
                  else
                  {
                      right = mid - 1;
                  }
              }
          }
          
          return -1;
      }
  };
  ```

- [第K小元素](https://leetcode.cn/problems/kth-smallest-element-in-a-sorted-matrix/)

  ```c++
  class Solution {
  public:
      int kthSmallest(vector<vector<int>>& matrix, int k) {
          // 二分查找
          int n = matrix.size();
          int left = matrix[0][0];
          int right = matrix[n-1][n-1];
  
          while(left < right)
          {
              int mid = left + (right - left) / 2;
  
              if(Count(matrix,n,mid) < k)
              {
                  left = mid + 1;
              }
              else
              {
                  right = mid;
              }
          }
  
          return left;
      }
  
      // 统计有序矩阵中小于等于x的数量
      int Count(vector<vector<int>>& matrix, int n, int num)
      {
          int res = 0;
          int col = 0;
          int row = n - 1;
  
          while(0 <= row && col < n)
          {
              if(matrix[row][col] <= num)
              {
                  res += (row + 1);
                  col++;
              }
              else
              {
                  row--;
              }
          }
  
          return res;
      }
  };
  ```
